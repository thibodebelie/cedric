import gurobipy as gp
from gurobipy import GRB
import pandas as pd

# --- 1. SETTINGS & PARAMETERS ---
STEPS = ["Step_1_Unload", "Step_2_Transport", "Step_3_Infeed"]
RESOURCES = {"Step_1_Unload": 4, "Step_2_Transport": 3, "Step_3_Infeed": 3}
SLACKS = {"S1_Limit": 10, "S2_Limit": 7, "S3_Limit": 10}
SCENARIO_FILE = "Bootstrapped_Baggage_Scenarios.csv"
BIG_M = 2000 # Larger than any possible time index

# --- 2. DATA LOADING ---
def get_stochastic_data():
    df_scenarios = pd.read_csv(SCENARIO_FILE)
    def to_min(ts):
        dt = pd.to_datetime(ts)
        return dt.hour * 60 + dt.minute - 360 # offset 6h
   
    scenarios_dict = {}
    all_flights = df_scenarios['Flight'].unique().tolist()
    for s_id in df_scenarios['Scenario_ID'].unique():
        s_df = df_scenarios[df_scenarios['Scenario_ID'] == s_id]
        scenarios_dict[s_id] = {row['Flight']: {
            "arrival": to_min(row['Ha Block']),
            "Step_1_Unload": row['Duration_S1'],
            "Step_2_Transport": row['Duration_S2'],
            "Step_3_Infeed": row['Duration_S3']
        } for _, row in s_df.iterrows()}
    return all_flights, scenarios_dict

FLIGHT_IDS, SCENARIOS = get_stochastic_data()
N_SCENARIOS = len(SCENARIOS)

# --- 3. MODEL FORMULATION ---
def solve_TSSP_BigM():
    model = gp.Model("TSSP_Inbound_BigM")
   
    # --- A. Variables ---
    # Stage 1: Machine Assignment (Scenario Independent)
    # y[f, s, m] = 1 if flight f uses machine m for step s
    y = model.addVars(FLIGHT_IDS, STEPS, range(1, 5), vtype=GRB.BINARY, name="assign")

    # Stage 2: Start Times (Scenario Dependent)
    st = {}
    for w in SCENARIOS:
        for f in SCENARIOS[w].keys():
            for s in STEPS:
                st[f, s, w] = model.addVar(lb=0, vtype=GRB.CONTINUOUS, name=f"st_{f}_{s}_{w}")

    # Binary variables for Big-M sequencing (Scenario Dependent)
    # z[f1, f2, s, w] = 1 if flight f1 processed before f2 in scenario w
    z = {}
    for w in SCENARIOS:
        present_flights = list(SCENARIOS[w].keys())
        for s in STEPS:
            for i, f1 in enumerate(present_flights):
                for f2 in present_flights[i+1:]:
                    z[f1, f2, s, w] = model.addVar(vtype=GRB.BINARY)

    # --- B. Constraints ---
   
    # 1. Each flight must be assigned to exactly one machine per step
    for f in FLIGHT_IDS:
        for s in STEPS:
            model.addConstr(y.sum(f, s, '*') == 1)
            # Ensure we don't assign to machine 4 if only 3 exist
            for m in range(RESOURCES[s] + 1, 5):
                model.addConstr(y[f, s, m] == 0)

    # 2. Capacity via Big-M Sequencing[cite: 1]
    for w in SCENARIOS:
        present_flights = list(SCENARIOS[w].keys())
        for s in STEPS:
            for i, f1 in enumerate(present_flights):
                for f2 in present_flights[i+1:]:
                    # If they share the same machine, one must start after the other ends[cite: 1]
                    # This only triggers if y[f1, s, m] == 1 AND y[f2, s, m] == 1
                    for m in range(1, RESOURCES[s] + 1):
                        # shared_machine is 1 if both flights f1 and f2 use machine m
                        shared_m = model.addVar(vtype=GRB.BINARY)
                        model.addConstr(shared_m >= y[f1, s, m] + y[f2, s, m] - 1)
                       
                        dur1 = SCENARIOS[w][f1][s]
                        dur2 = SCENARIOS[w][f2][s]
                       
                        # Big-M Sequencing Constraints[cite: 1]
                        model.addConstr(st[f1, s, w] + dur1 <= st[f2, s, w] + BIG_M * (1 - z[f1, f2, s, w]) + BIG_M * (1 - shared_m))
                        model.addConstr(st[f2, s, w] + dur2 <= st[f1, s, w] + BIG_M * (z[f1, f2, s, w]) + BIG_M * (1 - shared_m))

    # 3. Temporal & Slack Constraints[cite: 1]
    for w in SCENARIOS:
        for f, data in SCENARIOS[w].items():
            # Step 1 logic
            model.addConstr(st[f, "Step_1_Unload", w] >= data["arrival"])
            model.addConstr(st[f, "Step_1_Unload", w] <= data["arrival"] + SLACKS["S1_Limit"])
            # Step 2 logic
            model.addConstr(st[f, "Step_2_Transport", w] >= st[f, "Step_1_Unload", w] + data["Step_1_Unload"])
            model.addConstr(st[f, "Step_2_Transport", w] <= st[f, "Step_1_Unload", w] + data["Step_1_Unload"] + SLACKS["S2_Limit"])
            # Step 3 logic
            model.addConstr(st[f, "Step_3_Infeed", w] >= st[f, "Step_2_Transport", w] + data["Step_2_Transport"])
            model.addConstr(st[f, "Step_3_Infeed", w] <= st[f, "Step_2_Transport", w] + data["Step_2_Transport"] + SLACKS["S3_Limit"])

    # --- C. Objective ---
    scenario_avgs = []
    for w in SCENARIOS:
        n = len(SCENARIOS[w])
        if n > 0:
            sum_h = gp.quicksum(st[f, "Step_3_Infeed", w] + SCENARIOS[w][f]["Step_3_Infeed"] - SCENARIOS[w][f]["arrival"] for f in SCENARIOS[w])
            scenario_avgs.append(sum_h / n)
   
    model.setObjective(gp.quicksum(scenario_avgs) / N_SCENARIOS, GRB.MINIMIZE)
    model.setParam('TimeLimit', 600)
    model.optimize()

if __name__ == "__main__":
    solve_TSSP_BigM()
